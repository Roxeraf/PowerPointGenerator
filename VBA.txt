Private Sub KonfigLabel(ws As Worksheet, r As Long, label As String)
    ws.Rows(r).RowHeight = 26
    With ws.Cells(r, 2)
        .Value = label
        .Font.Bold = True
        .Font.Size = 10
        .Font.Color = RGB(0, 47, 94)
    End With
    ws.Range(ws.Cells(r, 2), ws.Cells(r, 7)).Interior.Color = RGB(200, 215, 235)
End Sub

Private Sub KonfigFeld(ws As Worksheet, r As Long, label As String, defaultVal As String)
    ws.Rows(r).RowHeight = 22
    With ws.Cells(r, 2)
        .Value = label
        .Font.Color = RGB(40, 60, 90)
    End With
    With ws.Cells(r, 3)
        .Value = defaultVal
        .Interior.Color = RGB(255, 255, 255)
        .Borders(xlEdgeBottom).LineStyle = xlContinuous
        .Borders(xlEdgeBottom).Color = RGB(0, 124, 193)
        .Borders(xlEdgeBottom).Weight = xlThin
        .Font.Color = RGB(0, 47, 94)
        .Font.Bold = True
        .Locked = False
    End With
End Sub

' 3. JSON-EXPORT + PPTX-GENERIERUNG
' =========================================================================
Sub Exportiere_Als_JSON_Und_PPTX()
    Dim wb As Workbook
    Dim wsKonfig As Worksheet
    Dim wsProj As Worksheet
    Dim headline As String
    Dim subheadline As String
    Dim kundenname As String
    Dim datum As String
    Dim ersteller As String
    Dim outputPfad As String
    Dim dateiname As String
    Dim nodePfad As String
    Dim skriptPfad As String
    Dim monate As Variant
    Dim projektJson As String
    Dim projName As String
    Dim json As String
    Dim jsonPfad As String
    Dim pptxAusgabe As String
    Dim befehl As String
    Dim logPfad As String
    Dim ersterProjekt As Boolean
    Dim i As Long
    Dim col As Long
    Dim fileNr As Integer
    Dim exitCode As Long
    Dim wshell As Object
    ' Teilprojekt-Variablen
    Dim tpStartRows As Variant
    Dim tpFteRows As Variant
    Dim tpArrayJson As String
    Dim tpPhasenJson As String
    Dim tpFteJson As String
    Dim tpName As String
    Dim tpKuerzel As String
    Dim tpCellVal As String
    Dim tpFteVal As Double
    Dim tpHeaderRow As Long
    Dim tpFteRow As Long
    Dim subRow As Long
    Dim tp As Long
    Dim firstTP As Boolean
    Dim firstTpPhase As Boolean
    Dim firstTpFte As Boolean
    Dim fteSumJson As String
    Dim firstFteSumme As Boolean
    Dim fteSumVal As Double

    Set wb = ActiveWorkbook
    Set wsKonfig = wb.Worksheets("Konfiguration")

    headline = Trim(wsKonfig.Cells(4, 3).Value)
    subheadline = Trim(wsKonfig.Cells(5, 3).Value)
    kundenname = Trim(wsKonfig.Cells(6, 3).Value)
    datum = Trim(wsKonfig.Cells(7, 3).Value)
    ersteller = Trim(wsKonfig.Cells(8, 3).Value)
    outputPfad = Trim(wsKonfig.Cells(11, 3).Value)
    dateiname = Trim(wsKonfig.Cells(12, 3).Value)
    nodePfad = Trim(wsKonfig.Cells(13, 3).Value)
    skriptPfad = Trim(wsKonfig.Cells(14, 3).Value)

    If Len(headline) = 0 Then headline = "Portfolio Kapazitaetsplanung"
    If Len(dateiname) = 0 Then dateiname = "PLX_Kapazitaetsplanung"
    If Len(nodePfad) = 0 Then nodePfad = "node"
    If Len(outputPfad) = 0 Then outputPfad = Environ("USERPROFILE") & "\Documents\"
    If Right(outputPfad, 1) <> "\" Then outputPfad = outputPfad & "\"

    monate = Array("Apr 26", "Mai 26", "Jun 26", "Jul 26", "Aug 26", "Sep 26", _
                   "Okt 26", "Nov 26", "Dez 26", "Jan 27", "Feb 27", "Mrz 27", "Apr 27", "Mai 27")

    ' Teilprojekt-Zeilen (Sub-Zeilen OHNE Header-Zeile)
    tpStartRows = Array(5, 11, 17)   ' Erste Sub-Zeile jedes TPs
    tpFteRows = Array(24, 25, 26)    ' FTE-Zeilen der drei TPs

    projektJson = "["
    ersterProjekt = True

    For i = 1 To 12
        Set wsProj = Nothing
        On Error Resume Next
        Set wsProj = wb.Worksheets("Projekt " & Format(i, "00"))
        On Error GoTo 0

        If Not wsProj Is Nothing Then

            ' Projektname aus Zelle A1 lesen
            projName = Trim(CStr(wsProj.Cells(1, 1).Value))
            If Len(projName) = 0 Then projName = "Projekt " & Format(i, "00")

            ' ---------------------------------------------------
            ' Teilprojekte
            ' ---------------------------------------------------
            tpArrayJson = "["
            firstTP = True

            For tp = 0 To 2
                tpHeaderRow = CLng(tpStartRows(tp)) - 1

                tpName = Trim(CStr(wsProj.Cells(tpHeaderRow, 1).Value))
                If Len(tpName) = 0 Then tpName = "Teilprojekt " & (tp + 1)

                ' Phasen
                tpPhasenJson = "{"
                firstTpPhase = True
                For col = 2 To 15
                    tpKuerzel = ""
                    For subRow = CLng(tpStartRows(tp)) To CLng(tpStartRows(tp)) + 4
                        tpCellVal = Trim(CStr(wsProj.Cells(subRow, col).Value))
                        If tpCellVal = "p" Or tpCellVal = "k" Or _
                           tpCellVal = "t" Or tpCellVal = "g" Or tpCellVal = "?" Then
                            tpKuerzel = tpCellVal
                            Exit For
                        End If
                    Next subRow
                    If Len(tpKuerzel) > 0 Then
                        If Not firstTpPhase Then tpPhasenJson = tpPhasenJson & ","
                        tpPhasenJson = tpPhasenJson & _
                            """" & EscapeJson(CStr(monate(CLng(col - 2)))) & """:" & _
                            """" & tpKuerzel & """"
                        firstTpPhase = False
                    End If
                Next col
                tpPhasenJson = tpPhasenJson & "}"

                ' FTE
                tpFteJson = "{"
                firstTpFte = True
                tpFteRow = CLng(tpFteRows(tp))
                For col = 2 To 15
                    tpFteVal = 0
                    On Error Resume Next
                    tpFteVal = CDbl(wsProj.Cells(tpFteRow, col).Value)
                    On Error GoTo 0
                    If tpFteVal > 0 Then
                        If Not firstTpFte Then tpFteJson = tpFteJson & ","
                        tpFteJson = tpFteJson & _
                            """" & EscapeJson(CStr(monate(CLng(col - 2)))) & """:" & _
                            Replace(CStr(tpFteVal), ",", ".")
                        firstTpFte = False
                    End If
                Next col
                tpFteJson = tpFteJson & "}"

                ' Nur TPs mit Daten exportieren
                If tpPhasenJson <> "{}" Or tpFteJson <> "{}" Then
                    If Not firstTP Then tpArrayJson = tpArrayJson & ","
                    tpArrayJson = tpArrayJson & _
                        "{""name"":""" & EscapeJson(tpName) & """," & _
                        """phasen"":" & tpPhasenJson & "," & _
                        """fte"":" & tpFteJson & "}"
                    firstTP = False
                End If
            Next tp
            tpArrayJson = tpArrayJson & "]"

            ' ---------------------------------------------------
            ' FTE-Summe (Zeile 27)
            ' ---------------------------------------------------
            fteSumJson = "{"
            firstFteSumme = True
            For col = 2 To 15
                fteSumVal = 0
                On Error Resume Next
                fteSumVal = CDbl(wsProj.Cells(27, col).Value)
                On Error GoTo 0
                If fteSumVal > 0 Then
                    If Not firstFteSumme Then fteSumJson = fteSumJson & ","
                    fteSumJson = fteSumJson & _
                        """" & EscapeJson(CStr(monate(CLng(col - 2)))) & """:" & _
                        Replace(CStr(fteSumVal), ",", ".")
                    firstFteSumme = False
                End If
            Next col
            fteSumJson = fteSumJson & "}"

            If Not ersterProjekt Then projektJson = projektJson & ","
            projektJson = projektJson & _
                "{""name"":""" & EscapeJson(projName) & """," & _
                """teilprojekte"":" & tpArrayJson & "," & _
                """fteSumme"":" & fteSumJson & "}"
            ersterProjekt = False
        End If
        Set wsProj = Nothing
    Next i
    projektJson = projektJson & "]"

    ' ---------------------------------------------------
    ' JSON zusammenbauen und schreiben
    ' ---------------------------------------------------
    json = "{" & vbNewLine
    json = json & "  ""titelseite"": {" & vbNewLine
    json = json & "    ""headline"": """ & EscapeJson(headline) & """," & vbNewLine
    json = json & "    ""subheadline"": """ & EscapeJson(subheadline) & """," & vbNewLine
    json = json & "    ""praesentationFuer"": """ & EscapeJson(kundenname) & """," & vbNewLine
    json = json & "    ""datum"": """ & EscapeJson(datum) & """," & vbNewLine
    json = json & "    ""ersteller"": """ & EscapeJson(ersteller) & """" & vbNewLine
    json = json & "  }," & vbNewLine
    json = json & "  ""projekte"": " & projektJson & vbNewLine
    json = json & "}"

    jsonPfad = outputPfad & dateiname & "_config.json"
    fileNr = FreeFile
    Open jsonPfad For Output As #fileNr
    Print #fileNr, json
    Close #fileNr

    pptxAusgabe = outputPfad & dateiname & ".pptx"

    If Len(skriptPfad) > 0 And skriptPfad <> "C:\PLX_Tools\PLX_generate_pptx.js" Then
        logPfad = outputPfad & "pptx_log.txt"
        befehl = nodePfad & " """ & skriptPfad & """ """ & jsonPfad & """ """ & pptxAusgabe & """"
        Set wshell = CreateObject("WScript.Shell")
        exitCode = wshell.Run( _
            "cmd /c " & befehl & " > """ & logPfad & """ 2>&1", _
            0, True)

        If exitCode = 0 Then
            MsgBox "PowerPoint erfolgreich erstellt!" & vbNewLine & vbNewLine & _
                   "Datei: " & pptxAusgabe, vbInformation, "Export erfolgreich"
            On Error Resume Next
            Shell "cmd /c start """" """ & pptxAusgabe & """", vbHide
            On Error GoTo 0
        Else
            MsgBox "Fehler beim Export (Exit: " & exitCode & ")." & vbNewLine & _
                   "JSON: " & jsonPfad & vbNewLine & _
                   "Log:  " & outputPfad & "pptx_log.txt" & vbNewLine & vbNewLine & _
                   "Pruefe Node.js Installation und Skript-Pfad.", _
                   vbExclamation, "Export fehlgeschlagen"
        End If
    Else
        MsgBox "JSON gespeichert:" & vbNewLine & vbNewLine & _
               jsonPfad & vbNewLine & vbNewLine & _
               "Fuehre dann manuell aus:" & vbNewLine & _
               "  node PLX_generate_pptx.js """ & jsonPfad & """ """ & pptxAusgabe & """" & vbNewLine & vbNewLine & _
               "Oder trage den Skript-Pfad im Konfigurationsblatt ein.", _
               vbInformation, "JSON exportiert"
    End If
End Sub

Private Function EscapeJson(s As String) As String
    s = Replace(s, "\", "\\")
    s = Replace(s, """", "\""")
    s = Replace(s, Chr(10), "\n")
    s = Replace(s, Chr(13), "")
    EscapeJson = s
End Function

' =========================================================================
' GANTT BEDINGTE FORMATIERUNG
' =========================================================================
Sub Setze_Gantt_Formatierung(rng As Range)
    rng.FormatConditions.Delete
    rng.HorizontalAlignment = xlCenter
    rng.VerticalAlignment = xlCenter
    With rng.FormatConditions.Add(Type:=xlCellValue, Operator:=xlEqual, Formula1:="=""p""")
        .Interior.Color = RGB(10, 85, 185):  .Font.Color = RGB(10, 85, 185)
    End With
    With rng.FormatConditions.Add(Type:=xlCellValue, Operator:=xlEqual, Formula1:="=""k""")
        .Interior.Color = RGB(40, 140, 225): .Font.Color = RGB(40, 140, 225)
    End With
    With rng.FormatConditions.Add(Type:=xlCellValue, Operator:=xlEqual, Formula1:="=""t""")
        .Interior.Color = RGB(195, 85, 215): .Font.Color = RGB(195, 85, 215)
    End With
    With rng.FormatConditions.Add(Type:=xlCellValue, Operator:=xlEqual, Formula1:="=""g""")
        .Interior.Color = RGB(205, 20, 90):  .Font.Color = RGB(205, 20, 90)
    End With
    With rng.FormatConditions.Add(Type:=xlCellValue, Operator:=xlEqual, Formula1:="=""?""")
        .Interior.Color = RGB(0, 47, 94):    .Font.Color = RGB(0, 47, 94)
    End With
End Sub

' =========================================================================
' FTE HEATMAP BEDINGTE FORMATIERUNG
' =========================================================================
Sub Setze_FTE_Formatierung(rng As Range)
    rng.FormatConditions.Delete
    rng.HorizontalAlignment = xlCenter
    rng.VerticalAlignment = xlCenter
    With rng.FormatConditions.Add(Type:=xlCellValue, Operator:=xlGreater, Formula1:="2")
        .Interior.Color = RGB(245, 200, 215): .Font.Color = RGB(150, 10, 60): .Font.Bold = True
    End With
    With rng.FormatConditions.Add(Type:=xlCellValue, Operator:=xlBetween, Formula1:="1.5", Formula2:="2")
        .Interior.Color = RGB(235, 205, 240): .Font.Color = RGB(120, 30, 140)
    End With
    With rng.FormatConditions.Add(Type:=xlCellValue, Operator:=xlBetween, Formula1:="1", Formula2:="1.49")
        .Interior.Color = RGB(200, 225, 245): .Font.Color = RGB(10, 85, 185)
    End With
    With rng.FormatConditions.Add(Type:=xlCellValue, Operator:=xlBetween, Formula1:="0.01", Formula2:="0.99")
        .Interior.Color = RGB(220, 245, 250): .Font.Color = RGB(20, 120, 160)
    End With
    With rng.FormatConditions.Add(Type:=xlCellValue, Operator:=xlEqual, Formula1:="0")
        .Font.Color = RGB(180, 190, 200)
    End With
End Sub
